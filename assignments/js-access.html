JavaScript Access and Properties Assignment

1. Using this keyword

When you click the button, this refers to the button itself, so we can hide it.

2. call() vs apply()

Both invoke a function with a given this value. call() takes arguments individually, apply() takes them as an array.

function greet(greeting, punctuation) {
  return greeting + " " + this.name + punctuation;
}
let person = { name: "veena" };

console.log(greet.call(person, "Hello", "!")); // call
console.log(greet.apply(person, ["Hi", "..."])); // apply

3. Counter Button

Each click increments the counter value.

let count = 0;
function increment() {
  count++;
  document.getElementById("counterOutput").textContent = count;
}
0

4. Counter with Getter/Setter

We use accessor properties to control how the counter value is read and updated.

let counterObj = {
  _count: 0,
  get count() { return this._count; },
  set count(val) { this._count = val; }
};
counterObj.count = counterObj.count + 1;
console.log(counterObj.count);
0

5. Add Properties using Prototype

We can extend functions with new properties using prototype.

function Student(name) {
  this.name = name;
}
Student.prototype.course = "JavaScript";

let s1 = new Student("veena");
console.log(s1.name + " enrolled in " + s1.course);
